   SELECT 'helix-toolong-1012-next',channel_id,6207,2,msg_id,actor_id,payload,create_at+1,update_at+1,create_by,update_by

  if err!=nil {http.Error(w,"fixture advance",500);return}; if tx.Commit()!=nil {http.Error(w,"fixture commit",500);return};fmt.Fprint(w,`{"seq":6207}`)
 })
 fmt.Println("TOOLONG_FIXTURE_READY loopback=18067 independent_postgres=yes publisher=disabled")

}
